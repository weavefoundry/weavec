# CMake embeds this file in the Core library (lib/Core/CMakeLists.txt) and
# core::LibrarySpec parses it on first use. unittests/Core/LibrarySpecTest.cpp
# checks every entry against an independent, hand-written expectation table,
# so an entry changed here must be changed there too.
#
# LEXICAL STRUCTURE
#   '#' starts a comment. A line whose last character before any comment is
#   '\' continues on the next line. Each logical line holds one directive or
#   one entry, ended by ';'. The file is ASCII.
#
# DIRECTIVES
#   header NAME;   NAME, relative to its include directory ("stdio.h",
#                  "sys/ioctl.h"), is a C library, POSIX or platform header
#                  (section 5.2), and the entries that follow are declared in
#                  it. "DIR/*" names every header under DIR; an entry cannot
#                  follow a pattern directly.
#   builtins;      The entries that follow are compiler builtins, declared by
#                  no header (__builtin_expect, __builtin_object_size).
#
#   A system header is a C library, POSIX or platform header when it is
#   under the toolchain's resource directory, when it belongs to a Darwin SDK
#   (every SDK header counts), or when it is named in a header directive.
#   Third-party libraries installed as system headers (jansson.h, sqlite3.h,
#   libcap's sys/capability.h) are not listed: their functions are unknown
#   callees (section 5.1).
#
# ENTRIES (section 8.2; '+' marks an extension of the RFC's grammar)
#   entry    ::= NAME '(' params ')' '->' result { clause } ';'
#   params   ::= [ param { ',' param } ] [ ',' '...' ]
#   param    ::= 'int' | 'other' | 'fn' { ':' fflag } | access { ':' pflag }
#   fflag    ::= callback | 'null-ok'+
#   access   ::= 'r' | 'w' | 'rw' | 'none'
#   pflag    ::= 'bytes(' term ')' | 'count(' term ')'+ | 'str' | 'null-ok'
#              | 'null-if-zero(' term ')' | 'release(' FAMILY ')'
#              | 'realloc(' FAMILY ')' | 'retain(' STATE ')' | 'escape'
#              | 'init(' FAMILY ')'+ | 'fini(' FAMILY ')'+
#              | 'out(' value ')'+ | callback+
#   callback ::= 'sync(' [ INT { ',' INT } ] ')' | 'entry(' [ INT ] ')'
#              | 'at-exit'
#   result   ::= 'void' | 'int' [ ':value(' term ')' ]+ | 'noreturn' | value
#   value    ::= rkind { ':' rflag }
#   rkind    ::= 'fresh(' FAMILY ')' | 'static(' STATE ')' | 'arg(' INT ')'
#              | 'interior(' INT ')' | 'interior-state(' STATE ')' | 'ptr'
#   rflag    ::= 'extent(' term ')' | 'null-on-failure' | 'nonnull'
#              | 'null-ok' | 'zero-init' | 'no-zero-init' | 'str'+
#              | 'replaces'+ | 'or-fresh(' FAMILY ')'+
#              | 'or-static(' STATE ')'+ | 'offset(' term ')'+
#              | 'zero-filled'+
#   clause   ::= 'disjoint(' INT ',' INT ',' term ')'
#              | 'copies(' INT ',' INT ',' term ')'+
#              | 'fills(' INT ',' term ',' term ')'+
#              | 'writes-str(' INT [ ',' term ] ')'+
#              | 'invalidates(' STATE ')' | 'reads(' STATE ')' | 'exits'
#              | 'returns-twice' | 'printf(' INT ',' INT ')'
#              | 'scanf(' INT ',' INT ')' | 'chk(' NAME ':' INT { ',' INT } ')'
#   term     ::= INT | 'a' N | 'strlen(a' N ')' | 'fmtlen(a' N ')' | MACRO+
#              | term '*' term | term '+' term | term '-' INT
#              | 'min(' term ',' term ')' | '(' term ')'+
#
# PARAMETERS
#   int, other, fn   an integer; any other non-pointer (floating, struct,
#                    va_list); a function pointer.
#   r, w, rw, none   the call reads, writes, reads and writes, or does not
#                    access the bytes behind the pointer. Without bytes, count
#                    or str the pointer needs one element of its pointee type.
#   bytes(t)         it needs t accessible bytes.
#   count(t)+        it needs t accessible elements of its pointee type
#                    (wchar_t for the wide functions).
#   str              its pointee holds a string terminated within its object
#                    (by a zero wchar_t for wide strings); needs r or rw.
#   null-ok          null is allowed. null-if-zero(t): allowed when t is zero
#                    (section 8.3). Otherwise null is forbidden.
#   release(F)       the call releases the pointer, of release family F.
#   realloc(F)       as realloc: released on a non-null result; on a null
#                    result kept, or released when the result's extent (the
#                    requested size) is zero; kept when there is no extent.
#   retain(S)        the pointer is kept in hidden state slot S; releasing it
#                    while kept makes the next reads(S) a use after free.
#   escape           the library keeps the pointer where no entry follows it
#                    (setvbuf's buffer, pthread_create's argument).
#   init(F)+         the object behind the pointer acquires a resource of
#                    family F (regcomp, glob). fini(F)+: that resource ends.
#                    The object's own storage is not the resource: it stays
#                    valid, fini on a local is no invalid release, and init
#                    after fini is no use after release.
#   out(v)+          the pointer points to a pointer slot, and the call stores
#                    v there (asprintf's buffer, strtol's end pointer). The
#                    nullability of v describes the stored value; with
#                    null-on-failure nothing usable is stored when the call
#                    reports failure.
#   sync(i, ...)     the function is called zero or more times before the call
#                    returns, its k-th pointer parameter pointing into call
#                    argument i_k (qsort: sync(0,0)); sync() when it only
#                    receives the library's own storage (nftw).
#   entry(i)         the function becomes an entry point (a thread, a signal
#                    handler) that receives argument i; entry() when it
#                    receives nothing from the call (signal).
#   at-exit          the function runs after the main flow has ended.
#                    On a pointer parameter, a callback flag describes the
#                    function pointers stored in the pointee (sigaction).
#
# RESULTS
#   void, int        no result; any non-pointer result.
#   noreturn         the call does not return.
#   fresh(F)         a new resource of family F: 'free' is the heap, 'stack'
#                    the calling frame (alloca); other families are named
#                    after their releaser (fclose, closedir, freeaddrinfo).
#   static(S)        a borrow of hidden state slot S, a global place <S>.
#   arg(N)           argument N itself.
#   interior(N)      a pointer into argument N's object. When argument N is
#                    null, the result points into the object that the entry's
#                    out(interior(N)) slot held before the call (strtok_r),
#                    or into static storage when there is no such slot
#                    (basename(NULL) is ".").
#   interior-state(S) a pointer into the value slot S retains (strtok).
#   ptr              a pointer of unknown provenance.
#   extent(t)        (fresh, static) the value has t accessible bytes.
#   nonnull, null-on-failure, null-ok
#                    never null (for arg: exactly the argument); null exactly
#                    when the call fails; possibly null. Defaults: fresh
#                    null-on-failure; static and arg nonnull; interior,
#                    interior-state and ptr null-ok.
#   zero-init, no-zero-init
#                    (section 11) whether a fresh value may be lowered to a
#                    zeroing wrapper; every fresh(free) value carries one. A
#                    str value without an extent was written up to its
#                    terminator.
#   str+             the value points to a terminated string.
#   replaces+        (out values) the slot's previous value is consumed as by
#                    realloc of the same family before the new one is stored.
#   or-fresh(F)+, or-static(S)+
#                    (arg results) when argument N is null the result is
#                    fresh(F) (realpath, getcwd) or static(S) (tmpnam).
#   value(t)+        (int results) the result equals t (strlen: strlen(a0)).
#   offset(t)+       (interior results) the result is t elements past the
#                    start of argument N (stpcpy: strlen(a1)).
#   zero-filled+     (fresh results) the library sets the whole extent to
#                    zero (calloc), whatever section 11 lowering does.
#
# CLAUSES
#   disjoint(a,b,t)  the first t bytes behind arguments a and b do not overlap.
#   copies(d,s,t)+   the call copies the first t bytes of argument s to
#                    argument d (memcpy; strncpy copies min(a2,strlen(a1)+1)).
#   fills(d,v,t)+    the call sets the first t bytes of argument d to v
#                    (memset: fills(0,a1,a2); bzero: fills(0,0,a1)).
#   writes-str(d[,t])+
#                    on success argument d holds a string of t characters,
#                    or of unknown length without t (fgets). Terms use the
#                    values before the call; a negative t writes nothing
#                    (snprintf with a zero size).
#   invalidates(S)   possibly ends every borrow of slot S (section 8.2),
#                    before the call's own result borrows S (getpwnam reuses
#                    and may reallocate its static buffer).
#   reads(S)         the call reads the values slot S retains.
#   exits            the call ends the process: leaks are not reported there
#                    (section 8.4). Needs noreturn.
#   returns-twice    setjmp-like (section 5.4).
#   printf(f,v), scanf(f,v)
#                    argument f is the format; v is the first variadic
#                    argument, or the va_list parameter of a v- function.
#   chk(A: i0, i1, ...)
#                    A is an alias of the entry (a fortified form): its k-th
#                    argument is the entry's argument i_k, or dropped (-1).
#                    For a variadic entry, alias arguments after the list
#                    continue from the last index. Every entry also answers to
#                    __builtin_NAME. The macOS SDK (secure/_string.h,
#                    secure/_stdio.h) and glibc (bits/*_fortified.h,
#                    bits/stdio2.h) spell the same builtins; glibc also calls
#                    the __NAME_chk functions directly.
#
# TERMS
#   aN               the value of argument N.
#   strlen(aN)       the length of argument N's string, in elements. Under
#                    min(t, e) a strlen(aN) in e is bounded by t, as
#                    strnlen(aN, t), and needs no terminator within t
#                    elements: min(a2,strlen(a1)+1) is what strncpy reads.
#   fmtlen(aN)       the length of the output of the format at argument N; a
#                    requirement, never a check term (section 8.1).
#   MACRO+           the value of that macro in the calling unit (BUFSIZ);
#                    unknown where it is not defined. No unit defines
#                    __WEAVEC_UNBOUNDED: a requirement that uses it is never
#                    proven or checked (gets).
#   Terms are mathematical integers; '*' binds tighter than '+' and '-'.
#
# OVERLOADS AND ABSENCES
#   A name may have several entries when their signatures (parameter classes,
#   arity, variadic, pointer result) differ; the engine uses the one that
#   accepts the declaration (glibc and BSD qsort_r, GNU and XSI strerror_r).
#   Arguments in '...' positions are borrowed for the call, and read or
#   written as the format says for printf and scanf entries.
#   Deliberately absent (section 5.2 applies): ioctl (its third argument's
#   layout depends on the request) and syscall. The buffers behind iovec and
#   msghdr arrays are nested in the argument, where no term can name them.
