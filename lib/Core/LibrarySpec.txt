# CMake embeds this file in the Core library (lib/Core/CMakeLists.txt) and
# core::LibrarySpec parses it on first use. The guard pass reads it
# (lib/Frontend/GuardLibrary.cpp, RFC 0035 section 2.5): before a direct call
# of a function named here, each pointer argument the row reads or writes is
# guarded over the bytes its term gives, evaluated on the call's arguments;
# a row with a wrapper is redirected to the runtime's checked version.
# unittests/Core/LibrarySpecTest.cpp checks the rows against an independent
# expectation table.
#
# LEXICAL STRUCTURE
#   '#' starts a comment. A line whose last character before any comment is
#   '\' continues on the next line. Each logical line holds one entry, ended
#   by ';'. The file is ASCII.
#
# ENTRIES
#   entry    ::= NAME '(' params ')' '->' result { clause } ';'
#   params   ::= [ param { ',' param } ] [ ',' '...' ]
#   param    ::= 'int' | 'other' | 'fn' | access { ':' pflag }
#   access   ::= 'r' | 'w' | 'rw' | 'none'
#   pflag    ::= 'bytes(' term ')' | 'count(' term ')' | 'str' | 'null-ok'
#              | 'null-if-zero(' term ')' | 'release' | 'realloc'
#              | 'scan' | 'scan(' term ')' | 'owner'
#   result   ::= 'void' | 'int' [ ':value(' term ')' ] | 'ptr' [ ':owned' ]
#   clause   ::= 'disjoint(' INT ',' INT ',' term ')' | 'wide'
#              | 'printf(' INT ',' INT ')' | 'scanf(' INT ',' INT ')'
#              | 'chk(' NAME ':' carg { ',' carg } ')' | 'wrapper(' NAME ')'
#              | 'exits' | 'unwinds'
#   carg     ::= INT | '_' | 'size'
#   term     ::= INT | 'a' N | 'strlen(a' N ')' | term '*' term
#              | term '+' term | term '-' INT | term '/' INT
#              | 'min(' term ',' term ')' | '(' term ')'
#
# PARAMETERS
#   int, other, fn   an integer; any other non-pointer (floating, struct,
#                    va_list); a function pointer. None is guarded.
#   r, w, rw, none   the call reads, writes, reads and writes, or does not
#                    access the bytes behind the pointer. Without bytes,
#                    count or str, its first byte is guarded.
#   bytes(t)         t bytes are guarded.
#   count(t)         (wide rows only) t wchar_t elements are guarded.
#   str              a string, guarded up to its terminator (a zero wchar_t
#                    in wide rows, which the guards do not scan).
#   scan(t)          (RFC 0038 section 2; read arguments of rows with a
#                    result and no wrapper) the call reads from the pointer
#                    up to the element where it stopped and t elements from
#                    there; scan alone is scan(1). A pointer result p stopped
#                    at element p - aN, an integer result n at element n; a
#                    null result means the whole extent was read: bytes(t)
#                    or count(t) when the row has one, else the string. The
#                    argument is guarded after the call, over those bytes;
#                    before it, only its null check and its first byte (none
#                    when bytes or count is zero).
#   null-ok          null is allowed. null-if-zero(t): allowed when t is
#                    zero. Otherwise null is forbidden.
#   release, realloc the call releases (free) or reallocates (realloc) the
#                    pointer: the allocator validates it, nothing is guarded.
#                    To Safe C (RFC 0043 section 2) the argument is an owner
#                    the call consumes (realloc: when its result is not null).
#   owner            the pointer points to a place the call stores an owner
#                    in (RFC 0043 section 2): asprintf's first argument.
#
# RESULTS
#   void, int, ptr   no result, any non-pointer result, a pointer.
#   ptr:owned        the caller owns the block the result points to (RFC
#                    0043 section 2); the allocations' blocks are zero.
#   value(t)         the result equals t (strlen: strlen(a0)); a call whose
#                    result only a value computes from what it reads is
#                    replaced by the guards that compute t.
#
# CLAUSES
#   disjoint(a,b,t)  the first t bytes behind arguments a and b must not
#                    overlap.
#   wide             the row's strings end in a zero wchar_t, and count(t)
#                    and strlen(aN) count wchar_t elements (wcslen).
#   printf(f,v), scanf(f,v)
#                    argument f is the format; v is the first variadic
#                    argument, or the va_list parameter of a v- function.
#                    A constant format's %s and %n arguments are guarded.
#   wrapper(N)       weavec-cc replaces the call with the runtime's checked
#                    __weavec_rt_N, which takes the call's arguments and its
#                    site (first when the call is variadic, last otherwise)
#                    and checks what the call reads and writes as it runs. N
#                    is the row's name, or its name and `_at` for a row
#                    that releases (RFC 0040 section 5.5); a C++ row's (a
#                    mangled name, RFC 0039 section 6) may name another.
#   exits, unwinds   (RFC 0039 section 3) the call ends the process, or
#                    unwinds the stack (a throw): the stack's shadow is not
#                    cleared before it, as it is before any other call that
#                    does not return.
#   chk(A: c0, c1, ...)
#                    A is an alias of the row (a fortified form): its k-th
#                    argument is the row's argument c_k, dropped ('_', a
#                    flag), or the size of the object the row writes
#                    ('size': the __builtin_object_size the header passed,
#                    in wchar_t elements for wide rows), which the guards
#                    check (RFC 0036 section 6). For a variadic row, alias
#                    arguments after the list continue from the last index.
#                    Every row also answers to __builtin_NAME. The macOS SDK
#                    (secure/_string.h, secure/_stdio.h) and glibc
#                    (bits/*_fortified.h, bits/stdio2.h) spell the same
#                    builtins; glibc also calls the __NAME_chk functions.
#
# TERMS
#   aN               the value of argument N.
#   strlen(aN)       the length of argument N's string, in elements. Under
#                    min(t, e) a strlen(aN) in e is bounded by t, as
#                    strnlen(aN, t), and needs no terminator within t
#                    elements: min(a2,strlen(a1)+1) is what strncpy reads.
#   t / N            t divided by the positive integer N, rounded down.
#   Terms are mathematical integers; a value that leaves 64 bits saturates.
#   '*' and '/' bind tighter than '+' and '-'.
#
# PORTABILITY
#   A row states what every supported C library (Darwin's libSystem, glibc)
#   does with an argument, not what a standard permits it to do: where they
#   differ, the row takes the more permissive behavior. Null is allowed if
#   any of them accepts it, and a count of zero allows null if any of them
#   then reads nothing. A row that requires more than a library does is a
#   false trap. An extent is what the call reads or writes, or, for a buffer
#   whose size is an argument, that size (the fortified forms of both
#   libraries enforce it). Element sizes are those of the 64-bit targets
#   (int and gid_t 4 bytes, pointers 8).
#
# C++
#   A C++ function's row is named by its Itanium mangling and lists the
#   arguments of its call in the IR: a member function's implicit object
#   first (RFC 0039 section 7).
#
# CLASSES
#   entry    ::= 'class' NAME { '::' NAME } ':' cclause { ',' cclause } ';'
#   cclause  ::= 'erased' | 'shared' | 'resets' '(' NAME { ',' NAME } ')'
#   RFC 0044. erased (section 2): a C++ class that can hold an object of a
#   type its own type does not name (a callable, a value of any type);
#   Safe C++ treats one made from a view as a view. shared (section 8): a
#   pointer whose object has other owners, which the function that holds
#   one does not see. resets (section 5):
#   member functions that leave the object in a known state whatever it
#   held, so that a moved-from object may be used again after one, as
#   after an assignment. Inline namespaces are not written (std::function,
#   not std::__1::function).
#
# OVERLOADS
#   A name may have several rows; a call uses the first whose arity it
#   matches (a variadic row accepts more arguments), or else the first.
#   Deliberately absent: ioctl (its third argument's layout depends on the
#   request) and syscall. The buffers behind iovec and msghdr arrays are
#   nested in the argument, where no term can name them.
